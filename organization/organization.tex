\section{Passing the Course}

\begin{frame}{Prerequisites}
    \begin{itemize}
        \item Basic foundations in economics and statistics 
        \item Successful completion of the \textbf{economics orientation module} (Economic Reasoning and Statistical Reasoning) 
        \item Or comparable coursework in a different university degree or at a different university
    \end{itemize}
\end{frame}

\begin{frame}{Active Participation}
    \begin{itemize}
        \item Attendance of at least 80\% of the classes in order to pass the course
        \item \textbf{Please be on time!}
        \item If you are unable to attend a class, let me know beforehand. Otherwise, I need a doctor's note
        \item Presentation at the end of the semester, regardless of participation in the final examination
    \end{itemize}
\end{frame}

\begin{frame}{Examination}
    \textbf{To pass the overall module, you need to attend two parts of the module, but you only need to take the examination in one of the courses.} \\
    For those who want to be graded in this part of the module, the final grade for the course will be a weighted average of two assignments:
    \begin{itemize}
        \item 25\% In-class presentation of an economic paper 
        \item 75\% Written exam
    \end{itemize}
\end{frame}

\begin{frame}{Presentation structure}
The presentation should last between 15-20 minutes.
    \begin{enumerate}
        \item Title page
        \item Motivation and research question
        \item Background
        \item Methodology
        \item Data 
        \item Main results 
        \item (Caveat, Limitations, Shortcomings)
        \item Conclusion, Policy implications
    \end{enumerate}
    A classroom discussion on the paper and topic will follow the presentation. Presenters should be prepared to answer questions related to the reading.
\end{frame}
